paths:
  /v1/ceding-carrier-verifications:
    post:
      operationId: createCedingCarrierVerification
      tags:
        - Ceding Carrier Verification
      summary: Verify ceding carrier policy affiliation
      description: >
        Verify that a sender (distributor) is affiliated with a ceding policy number at the
        ceding insurance carrier. The sender supplies the ceding policy number, its own sender
        identifier, and the last four digits of the annuitant ID on the policy; the ceding
        carrier confirms the match and returns the ceding carrier ID, processing location ID,
        CUSIP, and an indicator of whether the sender is affiliated with the policy. If the
        annuitant ID does not match the carrier's records for the referenced policy, the carrier
        returns an error instead of the verification resource.
      parameters:
        - $ref: '#/components/parameters/correlationId'
      requestBody:
        required: true
        content:
          application/json:
            schema:
              $ref: '#/components/schemas/CedingCarrierVerificationRequest'
      responses:
        '200':
          description: Ceding carrier verification successfully retrieved
          headers:
            correlationId:
              $ref: '#/components/headers/correlationId'
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/CedingCarrierVerification'
        '400': { $ref: '#/components/responses/BadRequestError' }
        '401': { $ref: '#/components/responses/UnauthorizedError' }

    CedingCarrierVerificationRequest:
      type: object
      description: Request to verify a sender's affiliation with a ceding policy number at the ceding insurance carrier.
      required:
        - policyNumber
        - senderId
        - annuitantId
      properties:
        policyNumber:
          type: string
          minLength: 1
          maxLength: 30
          pattern: '^[0-9A-Za-z]{1,30}$'
          description: This identifies the ceding policy number at the ceding insurance carrier when requesting ceding policy information.
        senderId:
          type: string
          minLength: 8
          maxLength: 8
          pattern: '^(?!0{8}$)[0-9]{8}$'
          description: This identifies the sender (distributor) requesting the ceding carrier information on a ceding policy that is potentially being surrendered.
        annuitantId:
          type: string
          minLength: 4
          maxLength: 4
          pattern: '^[0-9A-Za-z]{4}$'
          description: >
            Identifies the annuitant ID (last four digits) on the ceding policy. If the value
            does not match the ceding carrier's records, an error is returned to the sender
            instead of the verification resource.

    CedingCarrierVerification:
      type: object
      description: Result of verifying a sender's affiliation with a ceding policy number at the ceding insurance carrier.
      required:
        - policyNumber
        - annuitantId
        - cedingCarrierId
        - cedingProcessingLocationId
        - cusip
        - isDistributorAffiliated
      properties:
        policyNumber:
          type: string
          minLength: 1
          maxLength: 30
          pattern: '^[0-9A-Za-z]{1,30}$'
          description: This identifies the ceding policy number at the ceding insurance carrier when requesting ceding policy information. Echoed back from the request.
        annuitantId:
          type: string
          minLength: 4
          maxLength: 4
          pattern: '^[0-9A-Za-z]{4}$'
          description: Identifies the annuitant ID (last four digits) on the ceding policy. Echoed back from the request.
        cedingCarrierId:
          type: string
          minLength: 8
          maxLength: 8
          pattern: '^(?!0{8}$)[0-9]{8}$'
          description: >
            Identifies the DTCC Carrier ID that owns the policy and is reporting the ceding
            policy on the position and valuations (POV) file that is stored within IIEX.
        cedingProcessingLocationId:
          type: string
          minLength: 10
          maxLength: 10
          description: >
            This processing location ID identifies the correct location and processing
            attributes for the policy. This location also applies to 1035 exchange/replacement
            transactions as the location to direct information/processing attributes.
        cusip:
          type: string
          minLength: 9
          maxLength: 9
          pattern: '^[a-zA-Z0-9]{9}$'
          description: This field provides the name of the Policy-Product reported on the positions and valuations (POV) file that is stored within IIEX.
        isDistributorAffiliated:
          type: boolean
          description: This identifies if the sender (distributor) requesting ceding carrier information is affiliated with the ceding policy number.
